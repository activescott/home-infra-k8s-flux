# Plugins installed as managed copies by the install-plugins initContainer.
# Managed installs create the trust record required for openKeyedStore access
# (ACP session state). Pinned versions must match the OpenClaw version in the image.
#
# The npm: prefix is required from 2026.9.5, where a bare name in the official catalog
# installs from ClawHub into $OPENCLAW_STATE_DIR/extensions instead of the npm project
# layout the trust record and plugins.load.paths expect. Same fix as image/Dockerfile in
# activescott/activeassistant#249.
npm:@openclaw/acpx@2026.9.5

# A leading "-" removes that package's managed install from the PV. Slack now runs from

-npm:@openclaw/slack
